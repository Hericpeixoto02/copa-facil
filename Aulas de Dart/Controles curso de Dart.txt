Controles curso de Dart:


IF:

void main() {
  if (true) {
    print('imprimindo if');
  }

  print('Finalizou o if');

}


OU FALSE

void main() {
  if (false) {
    print('imprimindo if');
  }

  print('Finalizou o if');

}


OBS: Não imprime o primeiro IF


OPERADORES:


void main() {

  var sexo = 'masculino';
  if (sexo == 'masculino') {
    print('O sexo é masculino');
  }

  print('Finalizou o if');

}	




OPERADOR COM ELSE 

void main() {

  var idade = 17;
  if (idade > 18) {
    print('Héric pode ter cartão de crédito');
  }

else {
    print('Héric não pode ter cartão de crédito');
  }
  

}




OPERADOR ELASE IF:

void main() {

  var idade = 18;
  if (idade > 18) {
    print('Héric pode ter cartão de crédito');
  }

else if (idade >= 18) {
    print('Idade igual ou maior que 18 anos');
  }

else {
  print('Não pode ter cartão de crédito, menor de 18 anos');
};

}
