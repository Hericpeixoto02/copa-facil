While e do While - Estrutura de repetição:




WHILE:

void main(){
  int idade = 10;
  while (idade < 18){
    print(idade.toString() + ' é menor que 18 anos');
    idade ++;
  }
}


DO WHILE:

void main(){
  int idade = 10;
  while (idade < 18){
    print(idade.toString() + ' é menor que 18 anos');
    idade ++;
  }

  do {
  print('Teste do While');
} while (false);
}
